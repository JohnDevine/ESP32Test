1/**
2 * @file ESP32Test.c
3 * @brief Wi-Fi Access Point scanner for the ESP32.
4 *
5 * @details
6 * This application initializes the ESP-IDF Wi-Fi stack in station mode, performs a
7 * blocking scan for nearby access points, and logs information about every AP found
8 * (SSID, RSSI, authentication mode, cipher suites, and channel).
9 *
10 * The scan can optionally be limited to channels 1, 6, and 11 by defining
11 * @c CONFIG_EXAMPLE_USE_SCAN_CHANNEL_BITMAP. The maximum number of results stored
12 * is controlled by @c CONFIG_EXAMPLE_SCAN_LIST_SIZE (defaults to 10).
13 *
14 * @note This example is based on the ESP-IDF Wi-Fi scan example and is released
15 * into the Public Domain (or CC0 licensed, at your option).
16 */

187#ifdef USE_CHANNEL_BITMAP
188/**
189 * @brief Convert a channel list into a 2.4 GHz channel bitmap.
190 *
191 * @param[in] channel_list Array of 2.4 GHz channel numbers to enable.
192 * @param[in] channel_list_size Number of entries in @p channel_list.
193 * @param[out] scan_config Scan configuration whose bitmap is populated.
194 */
195static void array_2_channel_bitmap(const uint8_t channel_list[], const uint8_t channel_list_size, wifi_scan_config_t *scan_config) {
196
197 for(uint8_t i = 0; i < channel_list_size; i++) {
198 uint8_t channel = channel_list[i];
199 scan_config->channel_bitmap.ghz_2_channels |= (1 << channel);
200 }
201}
202#endif /*USE_CHANNEL_BITMAP*/

205/**
206 * @brief Initialize the Wi-Fi stack in station mode, scan, and log the results.
207 *
208 * Performs a blocking scan (or a channel-limited scan when
209 * @c CONFIG_EXAMPLE_USE_SCAN_CHANNEL_BITMAP is enabled) and prints up to
210 * @c DEFAULT_SCAN_LIST_SIZE access points.
211 */
212static void wifi_scan(void)
213{
214 ESP_ERROR_CHECK(esp_netif_init());
215 ESP_ERROR_CHECK(esp_event_loop_create_default());
216 esp_netif_t *sta_netif = esp_netif_create_default_wifi_sta();
217 assert(sta_netif);
218
219 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
220 ESP_ERROR_CHECK(esp_wifi_init(&cfg));
221
222 uint16_t number = DEFAULT_SCAN_LIST_SIZE;
223 wifi_ap_record_t ap_info[DEFAULT_SCAN_LIST_SIZE];
224 uint16_t ap_count = 0;
225 memset(ap_info, 0, sizeof(ap_info));
226
227
228 ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
229 ESP_ERROR_CHECK(esp_wifi_start());
230
231#ifdef USE_CHANNEL_BITMAP
232 wifi_scan_config_t *scan_config = (wifi_scan_config_t *)calloc(1,sizeof(wifi_scan_config_t));
233 if (!scan_config) {
234 ESP_LOGE(TAG, "Memory Allocation for scan config failed!");
235 return;
236 }
237 array_2_channel_bitmap(channel_list, CHANNEL_LIST_SIZE, scan_config);
238 esp_wifi_scan_start(scan_config, true);
239 free(scan_config);
240
241#else
242 esp_wifi_scan_start(NULL, true);
243#endif /*USE_CHANNEL_BITMAP*/
244
245 ESP_LOGI(TAG, "Max AP number ap_info can hold = %u", number);
246 ESP_ERROR_CHECK(esp_wifi_scan_get_ap_num(&ap_count));
247 ESP_ERROR_CHECK(esp_wifi_scan_get_ap_records(&number, ap_info));
248 ESP_LOGI(TAG, "Total APs scanned = %u, actual AP number ap_info holds = %u", ap_count, number);
249 for (int i = 0; i < number; i++) {
250 ESP_LOGI(TAG, "SSID \t\t%s", ap_info[i].ssid);
251 ESP_LOGI(TAG, "RSSI \t\t%d", ap_info[i].rssi);
252 print_auth_mode(ap_info[i].authmode);
253 if (ap_info[i].akm_dpp) {
254 ESP_LOGI(TAG, "DPP \t\tSupported%s", (ap_info[i].authmode != WIFI_AUTH_DPP) ? " (mixed mode)" : " (DPP-only)");
255 }
256 if (ap_info[i].authmode != WIFI_AUTH_WEP) {
257 print_cipher_type(ap_info[i].pairwise_cipher, ap_info[i].group_cipher);
258 }
259 ESP_LOGI(TAG, "Channel \t\t%d", ap_info[i].primary);
260 }
261}

263/**
264 * @brief Application entry point.
265 *
266 * Initializes NVS (erasing and retrying if the partition is full or from an
267 * incompatible version) and then starts the Wi-Fi scan.
268 */
269void app_main(void)
270{
271 // Initialize NVS
272 esp_err_t ret = nvs_flash_init();
273 if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
274 ESP_ERROR_CHECK(nvs_flash_erase());
275 ret = nvs_flash_init();
276 }
277 ESP_ERROR_CHECK( ret );
278
279 wifi_scan();
280}
